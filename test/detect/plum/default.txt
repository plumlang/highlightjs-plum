seed math

root GREETING = "Hello"

# A planet, and the area of its surface.
tree Planet {
    name: str
    radius_km: float

    branch area() -> float => 4.0 * math.PI * self.radius_km ** 2
}

branch describe(p: Planet) -> str {
    return "{GREETING}, {p.name}! {p.area():.0} km²"
}

plant planets = [Planet("Mercury", 2439.7), Planet("Earth", 6371.0)]
for p in planets {
    drop(describe(p))
}
