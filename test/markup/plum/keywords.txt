    drop("many")
} else if count in [1, 2] {
    drop("few")
} else {
    drop("none")
}

    if i == 1 {
        continue
    }
    break
}

while count > 0 {
    count -= 1
}

    "start" => drop("starting")
    "stop", "halt" => drop("stopping")

}
